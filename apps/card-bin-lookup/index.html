# 批量查询示例

Python 3 接入代码
import requests

headers = {"X-Access-Password": "你的密码"}
res = requests.get("https://bin.11231632.xyz/api/lookup", params={"bin": "4246711234"}, headers=headers).json()
if res.get("success"):
    card = res["data"]
    print(f"发卡行: {card['issuer']}")
    print(f"国家: {card['country']} ({card['currency_code']})")
    print(f"品牌: {card['card_brand']} | 级别: {card['card_category']}")
